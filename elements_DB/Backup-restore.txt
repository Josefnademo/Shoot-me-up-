--Backup de la base de données
--Pour sauvegarder votre base de données jeu_database, utilisez :

mysqldump -u username -p jeu_database > backup_shoot_me_up.sql



--Restore de la base de données
--Pour restaurer la base de données à partir de la sauvegarde, utilisez :

mysql -u username -p jeu_database < backup_shoot_me_up.sql
